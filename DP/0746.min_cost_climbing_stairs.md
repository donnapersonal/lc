# 746.Min Cost Climbing Stairs

难度: Easy
标签: Array、DP

![746](https://github.com/donnapersonal/picx-images-hosting/raw/master/image.1hsy6w6avo.webp)

限制：
- 2 <= cost.length <= 1000
- 0 <= cost[i] <= 999

## 解题

```python
# 写法 1
class Solution:
    def minCostClimbingStairs(self, cost: List[int]) -> int:
        n = len(cost)
        dp = [0] * (n + 1)
        for i in range(2, n + 1):
            dp[i] = min(dp[i - 1] + cost[i - 1], dp[i - 2] + cost[i - 2])

        return dp[n]

# 写法 2
class Solution:
    def minCostClimbingStairs(self, cost: List[int]) -> int:
        n = len(cost)
        prev, cur = 0, 0  # 初始化 dp[0] 和 dp[1]
    
        for i in range(2, n + 1):
            temp = min(cur + cost[i - 1], prev + cost[i - 2])
            prev, cur = cur, temp  # 更新状态
        
        return cur
```
```java
// 写法 1
class Solution {
    public int minCostClimbingStairs(int[] cost) {
        int n = cost.length;
        int[] dp = new int[n+1];
        for (int i = 2; i < n+1; i++) {
            dp[i] = Math.min(dp[i-1] + cost[i-1], dp[i-2] + cost[i-2]);
        }
        return dp[n];
    }
}

// 写法 2
class Solution {
    public int minCostClimbingStairs(int[] cost) {
        int n = cost.length;
        int prev = 0, cur = 0;
        for (int i = 2; i < n+1; i++) {
            int temp = Math.min(cur + cost[i-1], prev + cost[i-2]);
            prev = cur;
            cur = temp;
        }
        return cur;
    }
}
```

- 时间复杂度：`O(n)`
- 空间复杂度：`O(1)`
